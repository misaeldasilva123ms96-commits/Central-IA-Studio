cmake_minimum_required(VERSION 3.19)
project(CentralEngine VERSION 0.3.0 LANGUAGES C CXX)
set(CENTRAL_LLAMA_COMMIT "d1044bc82712fbf08996080bef6298b24a27e9c7")
find_package(Python3 REQUIRED COMPONENTS Interpreter)
if(CENTRAL_LLAMA_SOURCE)
    set(llama_SOURCE_DIR "${CENTRAL_LLAMA_SOURCE}")
else()
    set(llama_SOURCE_DIR "${CMAKE_BINARY_DIR}/upstream-src")
    execute_process(COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/scripts/fetch_upstream.py ${llama_SOURCE_DIR} RESULT_VARIABLE fetch_result)
    if(NOT fetch_result EQUAL 0)
        message(FATAL_ERROR "Falha ao obter a fonte upstream fixada")
    endif()
endif()
execute_process(COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/scripts/prepare_upstream.py ${llama_SOURCE_DIR} RESULT_VARIABLE patch_result)
if(NOT patch_result EQUAL 0)
    message(FATAL_ERROR "Falha ao verificar e preparar llama.cpp")
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>" CACHE STRING "" FORCE)
foreach(flag GGML_CUDA GGML_VULKAN GGML_METAL GGML_SYCL GGML_RPC GGML_OPENMP GGML_NATIVE LLAMA_OPENSSL LLAMA_BUILD_TESTS LLAMA_BUILD_EXAMPLES LLAMA_BUILD_UI LLAMA_USE_PREBUILT_UI LLAMA_CONNECT)
    set(${flag} OFF CACHE BOOL "" FORCE)
endforeach()
foreach(flag GGML_CPU GGML_AVX GGML_AVX2 GGML_FMA GGML_F16C LLAMA_BUILD_COMMON LLAMA_BUILD_TOOLS LLAMA_BUILD_SERVER)
    set(${flag} ON CACHE BOOL "" FORCE)
endforeach()
set(GGML_AVX512 OFF CACHE BOOL "" FORCE)
add_subdirectory(${llama_SOURCE_DIR} ${CMAKE_BINARY_DIR}/llama EXCLUDE_FROM_ALL)
target_include_directories(llama-server-impl PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/src)
add_executable(central-engine-nocuda src/main.cpp)
target_compile_features(central-engine-nocuda PRIVATE cxx_std_17)
target_link_libraries(central-engine-nocuda PRIVATE llama-server-impl)
set_target_properties(central-engine-nocuda PROPERTIES RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin)
install(TARGETS central-engine-nocuda RUNTIME DESTINATION .)
